function handleFiles(files) { if (files.length === 0) return; const file = files[0]; if (!file.name.endsWith('.musicxml') && !file.name.endsWith('.xml')) { showToast('Invalid file. Use .musicxml', 'error'); return; } showToast('Processing score...', 'info'); const reader = new FileReader(); reader.onload = async function(e) { currentXmlData = e.target.result; try { // STEP 1: Unhide the container FIRST so it has a physical width document.getElementById('drop-zone').classList.add('hidden-view'); document.getElementById('osmd-host-wrapper').classList.remove('hidden-view'); // STEP 2: Load and Render the music await osmdHost.load(currentXmlData); osmdHost.render(); showToast('Score rendered on screen.', 'success'); } catch (err) { console.error("OSMD Render Error:", err); showToast('Failed to draw sheet music.', 'error'); return; } // STEP 3: Upload to Firebase if (isFirebaseActive && roomCode) { setTimeout(() => { showToast('Uploading score to sync engine...', 'info'); set(ref(db, `rooms/${roomCode}/xmlData`), currentXmlData) .then(() => showToast('Score ready for students!', 'success')) .catch((err) => { console.error("Firebase Upload Error:", err); showToast('Cloud upload failed.', 'error'); }); }, 500); } }; reader.readAsText(file); }
